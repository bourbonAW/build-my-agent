> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Flywheel control-surface UI: a React + TypeScript + Vite app with a typed API client, the MVP routes (runs, data/error analysis, taxonomy, judges, baselines, issues, proposals, regressions, settings) plus Phase-2 routes, decision forms for the human gates, and a Playwright test covering the main control loop. **Architecture:** `flywheel/ui/` Vite app. `src/api/` holds a typed client + TanStack Query hooks; `src/lib/lifecycle.ts` mirrors the authoritative enums + `deriveRegressionStatus` from plan 02; `src/routes/` holds pages; shared primitives in `src/components/`. **Tech Stack:** React 18, TypeScript, Vite, React Router, TanStack Query, TanStack Table, shadcn/ui (or local equivalents), lucide-react, Recharts, Vitest + Testing Library + Playwright.

## Global Constraints

(See `2026-06-23-flywheel-00-index.md`.) Most relevant here:
- **Authoritative lifecycle types verbatim**; `RegressionStatus` is **derived** from `ProposalState` (mirror plan 02 `derive_regression_status`).
- **Browser never receives Langfuse write credentials.** All writes go through Flywheel API.
- **No hidden automation**: LLM-generated labels/root-causes/proposals must show source, version, confidence.
- **Evidence before action**: every approve/reject/publish view must show traces, cases, labels, judge validity, holdout integrity, regression deltas.
- **Noise is a state**: `no_significant_change` is distinct from win/loss.
- **Human gates**: publish/rollback/revert/approve require explicit harness-owner action; UI disables actions the role can't perform.
- **Do not clone Langfuse**: deep trace inspection deep-links to Langfuse; redacted evidence only in-app.

```typescript
// flywheel/ui/src/lib/lifecycle.ts
// Authoritative lifecycle types. Mirrors plan 02 api/lifecycle.py — single source of truth.
export type ProposalState =
  | "draft" | "under_review" | "rejected" | "deferred" | "approved"
  | "handoff_ready" | "implementing" | "diff_review" | "revising"
  | "regression_running" | "regression_review" | "blocked_on_judge_recheck"
  | "blocked_on_judge_migration" | "baseline_stale" | "validated"
  | "rolled_back" | "no_significant_change" | "abandoned";

export type RegressionStatus =
  | "not_started" | "running" | "waiting_for_judge_recheck"
  | "waiting_for_judge_migration" | "ready_for_review" | "complete";

export type RegressionOutcome =
  | "published" | "rolled_back" | "no_significant_change" | "revise"
  | "abandoned" | "judge_recheck_required" | "judge_migration_required"
  | "baseline_stale";

export type RunState =
  | "idle" | "collecting" | "scored" | "waiting_for_labels" | "labels_synced"
  | "manual_analysis_ready" | "auto_analysis_ready" | "analyzing"
  | "clean_run" | "needs_more_data" | "under_review";

export type JudgeState =
  | "draft" | "calibrating" | "locked_test" | "validated"
  | "validated_limited" | "rejected" | "recheck_required";

const NOT_STARTED = new Set<ProposalState>([
  "draft", "under_review", "approved", "handoff_ready", "implementing", "diff_review",
]);
const COMPLETE = new Set<ProposalState>([
  "validated", "rolled_back", "no_significant_change", "abandoned",
]);

export function deriveRegressionStatus(state: ProposalState): RegressionStatus | null {
  if (NOT_STARTED.has(state)) return "not_started";
  if (state === "regression_running") return "running";
  if (state === "blocked_on_judge_recheck") return "waiting_for_judge_recheck";
  if (state === "blocked_on_judge_migration") return "waiting_for_judge_migration";
  if (state === "regression_review") return "ready_for_review";
  if (COMPLETE.has(state)) return "complete";
  // baseline_stale, revising, deferred, rejected -> omit
  return null;
}
```

> Note: `JudgeState` retains `locked_test` here because UI §12 lists it as a display abstraction; the DB/API enum (plan 02) omits it. UI must treat `locked_test` as display-only and never POST it as a judge status.

```typescript
// flywheel/ui/src/api/client.ts
export class ApiError extends Error {
  status: number;
  isUnauthorized: boolean;
  detail: string;
  constructor(status: number, detail: string) {
    super(`API ${status}: ${detail}`);
    this.status = status;
    this.detail = detail;
    this.isUnauthorized = status === 403;
  }
}

async function parse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    let detail = res.statusText;
    try {
      const body = await res.json();
      detail = body.detail ?? detail;
    } catch { /* non-JSON body */ }
    throw new ApiError(res.status, detail);
  }
  return (await res.json()) as T;
}

export async function apiGet<T>(path: string): Promise<T> {
  return parse<T>(await fetch(path, { method: "GET" }));
}

export async function apiPost<T>(
  path: string, body: unknown, opts?: { idempotencyKey?: string },
): Promise<T> {
  const headers: Record<string, string> = { "Content-Type": "application/json" };
  if (opts?.idempotencyKey) headers["Idempotency-Key"] = opts.idempotencyKey;
  return parse<T>(await fetch(path, {
    method: "POST", headers, body: JSON.stringify(body),
  }));
}
```

```typescript
function useGateMutation(path: (id: string) => string, invalidateKeys: (vars: {
  id: string; project: string;
}) => unknown[][], keyParts: (vars: {
  id: string; project: string; extra?: Record<string, unknown>;
}) => string[] = () => []) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (vars: { id: string; project: string; extra?: Record<string, unknown> }) =>
      apiPost(path(vars.id), { project: vars.project, ...(vars.extra ?? {}) },
        { idempotencyKey: [path(vars.id), vars.project, ...keyParts(vars)].join(":") }),
    onSuccess: (_d, vars) => {
      for (const queryKey of invalidateKeys(vars)) {
        qc.invalidateQueries({ queryKey });
      }
    },
  });
}
```

```tsx
export function ErrorState({ error }: { error: unknown }) {
  if (!(error instanceof ApiError)) {
    return <p role="alert">Something went wrong.</p>;

  if (error.status === 403) return <p role="alert">Action unavailable: {error.detail}</p>;
  if (error.status === 404) return <p role="alert">Not found: {error.detail}</p>;
  if (error.status === 409) return <p role="alert">Conflict: {error.detail}</p>;
  if (error.status === 503) return <p role="alert">Temporarily unavailable: {error.detail}</p>;
```

```tsx
export type Role = "dataset_curator" | "judge_owner" | "harness_owner" | "platform_maintainer";

export function RoleGate(
  { roles, required, children }:
  { roles: Set<Role>; required: Role; children: ReactElement<{ disabled?: boolean; title?: string }> },
) {
  const allowed = roles.has(required);
  if (!isValidElement(children)) return children;
  return cloneElement(children, {
    disabled: children.props.disabled || !allowed,
    title: allowed ? children.props.title : `Requires ${required}`,
  });
}
```

```tsx
// flywheel/ui/src/routes/ProposalReview.tsx
import { useState } from "react";
import {
  useApproveProposal, useCreateHandoff, useDeferProposal,
  useImplementationLink, useProposal, useRejectProposal,
} from "../api/hooks";
import type { Role } from "../components/RoleGate";
import { ErrorState } from "../components/ErrorState";
import { EvidencePanel } from "../components/EvidencePanel";
import { RoleGate } from "../components/RoleGate";
import { StatusBadge } from "../components/StatusBadge";

export function ProposalReview(
  { proposalId, project, roles, langfuseBaseUrl }:
  { proposalId: string; project: string; roles: Set<Role>; langfuseBaseUrl: string },
) {
  const { data: p, isLoading } = useProposal(proposalId);
  const approve = useApproveProposal();
  const reject = useRejectProposal();
  const defer = useDeferProposal();
  const handoff = useCreateHandoff();
  const link = useImplementationLink();
  const [prUrl, setPrUrl] = useState("");
  if (isLoading || !p) return <p>Loading proposal…</p>;
  const risk = p.proposed_changes.some((c) => c.risk_level === "high") ? "high"

  const mutationError = approve.error ?? reject.error ?? defer.error ?? handoff.error ?? link.error;
  return (

      <h1>Proposal {p.id} <StatusBadge kind="proposal" value={p.state} /></h1>
      {mutationError && <ErrorState error={mutationError} />}
      <p>Risk: {risk}</p>
      <p>Candidate hypothesis: {p.candidate_hypothesis_id}</p>
      <h2>Linked issues</h2>

      ))}

          <strong>{c.target_file}</strong> ({c.change_type}, risk {c.risk_level})
          <p>{c.description}</p>
          <pre>{c.suggested_diff}</pre>
        </li>
      ))}</ul>
      <h2>Expected metric delta</h2>
      <ul>{Object.entries(p.expected_metric_delta).map(([k, v]) =>
        <li key={k}>{k}: {v}</li>)}</ul>
      <h2>Rollback plan</h2>
      <p>{p.rollback_plan}</p>
      <div role="group" aria-label="decision">
        <RoleGate roles={roles} required="harness_owner">
          <button onClick={() => approve.mutate({ id: p.id, project })}>Approve</button>
        </RoleGate>
        <RoleGate roles={roles} required="harness_owner">
          <button onClick={() => reject.mutate({ id: p.id, project })}>Reject</button>
        </RoleGate>
        <RoleGate roles={roles} required="harness_owner">
          <button onClick={() => defer.mutate({ id: p.id, project })}>Defer</button>
        </RoleGate>
        <RoleGate roles={roles} required="harness_owner">
          <button onClick={() => handoff.mutate({ id: p.id, project })}>Generate handoff</button>

      }}>

        </RoleGate>
      </form>
    </section>
  );
}
```

```tsx
  { rows, columns }: { rows: T[]; columns: { key: string; header: string;

    <table>

      <tbody>

        ))}
```
