# Flywheel 08 — Web UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Flywheel control-surface UI: a React + TypeScript + Vite app with a typed API client, the MVP routes (runs, data/error analysis, taxonomy, judges, baselines, issues, proposals, regressions, settings) plus Phase-2 routes, decision forms for the human gates, and a Playwright test covering the main control loop. The browser talks only to Flywheel API and never holds Langfuse write credentials.

**Architecture:** `flywheel/ui/` Vite app. `src/api/` holds a typed client + TanStack Query hooks; `src/lib/lifecycle.ts` mirrors the authoritative enums + `deriveRegressionStatus` from plan 02; `src/routes/` holds pages; shared primitives in `src/components/`. Server state via TanStack Query; dense tables via TanStack Table. Vitest + Testing Library for components; Playwright for the e2e loop.

**Tech Stack:** React 18, TypeScript, Vite, React Router, TanStack Query, TanStack Table, shadcn/ui (or local equivalents), lucide-react, Recharts, Vitest + Testing Library + Playwright.

## Global Constraints

(See `2026-06-23-flywheel-00-index.md`.) Most relevant here:
- **Authoritative lifecycle types verbatim**; `RegressionStatus` is **derived** from `ProposalState` (mirror plan 02 `derive_regression_status`).
- **Browser never receives Langfuse write credentials.** All writes go through Flywheel API.
- **No hidden automation**: LLM-generated labels/root-causes/proposals must show source, version, confidence.
- **Evidence before action**: every approve/reject/publish view must show traces, cases, labels, judge validity, holdout integrity, regression deltas.
- **Noise is a state**: `no_significant_change` is distinct from win/loss.
- **Human gates**: publish/rollback/revert/approve require explicit harness-owner action; UI disables actions the role can't perform.
- **Do not clone Langfuse**: deep trace inspection deep-links to Langfuse; redacted evidence only in-app.

```typescript
// flywheel/ui/src/lib/lifecycle.ts
// Authoritative lifecycle types. Mirrors plan 02 api/lifecycle.py — single source of truth.
export type ProposalState =
  | "draft" | "under_review" | "rejected" | "deferred" | "approved"
  | "handoff_ready" | "implementing" | "diff_review" | "revising"
  | "regression_running" | "regression_review" | "blocked_on_judge_recheck"
  | "blocked_on_judge_migration" | "baseline_stale" | "validated"
  | "rolled_back" | "no_significant_change" | "abandoned";

export type RegressionStatus =
  | "not_started" | "running" | "waiting_for_judge_recheck"
  | "waiting_for_judge_migration" | "ready_for_review" | "complete";

export type RegressionOutcome =
  | "published" | "rolled_back" | "no_significant_change" | "revise"
  | "abandoned" | "judge_recheck_required" | "judge_migration_required"
  | "baseline_stale";

export type RunState =
  | "idle" | "collecting" | "scored" | "waiting_for_labels" | "labels_synced"
  | "manual_analysis_ready" | "auto_analysis_ready" | "analyzing"
  | "clean_run" | "needs_more_data" | "under_review";

export type JudgeState =
  | "draft" | "calibrating" | "locked_test" | "validated"
  | "validated_limited" | "rejected" | "recheck_required";

const NOT_STARTED = new Set<ProposalState>([
  "draft", "under_review", "approved", "handoff_ready", "implementing", "diff_review",
]);
const COMPLETE = new Set<ProposalState>([
  "validated", "rolled_back", "no_significant_change", "abandoned",
]);

export function deriveRegressionStatus(state: ProposalState): RegressionStatus | null {
  if (NOT_STARTED.has(state)) return "not_started";
  if (state === "regression_running") return "running";
  if (state === "blocked_on_judge_recheck") return "waiting_for_judge_recheck";
  if (state === "blocked_on_judge_migration") return "waiting_for_judge_migration";
  if (state === "regression_review") return "ready_for_review";
  if (COMPLETE.has(state)) return "complete";
  // baseline_stale, revising, deferred, rejected -> omit
  return null;
}
```

> Note: `JudgeState` retains `locked_test` here because UI §12 lists it as a display abstraction; the DB/API enum (plan 02) omits it. UI must treat `locked_test` as display-only and never POST it as a judge status.

```typescript
// flywheel/ui/src/api/client.ts
export class ApiError extends Error {
  status: number;
  isUnauthorized: boolean;
  detail: string;
  constructor(status: number, detail: string) {
    super(`API ${status}: ${detail}`);
    this.status = status;
    this.detail = detail;
    this.isUnauthorized = status === 403;
  }
}

async function parse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    let detail = res.statusText;
    try {
      const body = await res.json();
      detail = body.detail ?? detail;
    } catch { /* non-JSON body */ }
    throw new ApiError(res.status, detail);
  }
  return (await res.json()) as T;
}

export async function apiGet<T>(path: string): Promise<T> {
  return parse<T>(await fetch(path, { method: "GET" }));
}

export async function apiPost<T>(
  path: string, body: unknown, opts?: { idempotencyKey?: string },
): Promise<T> {
  const headers: Record<string, string> = { "Content-Type": "application/json" };
  if (opts?.idempotencyKey) headers["Idempotency-Key"] = opts.idempotencyKey;
  return parse<T>(await fetch(path, {
    method: "POST", headers, body: JSON.stringify(body),
  }));
}
```

```tsx
// flywheel/ui/src/components/StatusBadge.tsx
type Tone = "red" | "green" | "amber" | "blue" | "gray";

const TONE: Record<string, Tone> = {
  // green
  validated: "green", published: "green", current: "green", active: "green",
  // red
  rejected: "red", rolled_back: "red", reverted: "red", abandoned: "red",
  blocked_on_judge_recheck: "red", blocked_on_judge_migration: "red",
  recheck_required: "red", baseline_stale: "red",
  // amber
  under_review: "amber", deferred: "amber", revising: "amber",
  no_significant_change: "amber", calibrating: "amber", validated_limited: "amber",
  waiting_for_judge_recheck: "amber", waiting_for_judge_migration: "amber",
  ready_for_review: "amber",
  // blue
  collecting: "blue", analyzing: "blue", running: "blue", regression_running: "blue",
  implementing: "blue",
};

const TONE_COLOR: Record<Tone, string> = {
  red: "#b91c1c", green: "#15803d", amber: "#b45309", blue: "#1d4ed8", gray: "#6b7280",
};

export function StatusBadge({ kind, value }: { kind: string; value: string }) {
  const tone = TONE[value] ?? "gray";
  const label = value === "no_significant_change" ? "no significant change" : value;
  return (
    <span data-tone={tone} title={`${kind}: ${value}`}
      style={{ color: TONE_COLOR[tone], border: `1px solid ${TONE_COLOR[tone]}`,
               borderRadius: 4, padding: "1px 6px", fontSize: 12 }}>
      {label}
    </span>
  );
}
```

```typescript
// flywheel/ui/src/api/hooks.ts
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { apiGet, apiPost } from "./client";
import type { Run, Baseline, ProposalReview, RegressionResult } from "./types";

export function useRuns(project: string) {
  return useQuery({
    queryKey: ["runs", project],
    queryFn: () => apiGet<{ runs: Run[] }>(`/api/runs?project=${project}`),
  });
}

export function useBaselines(project: string) {
  return useQuery({
    queryKey: ["baselines", project],
    queryFn: () => apiGet<{ baselines: Baseline[] }>(`/api/baselines?project=${project}`),
  });
}

export function useProposal(id: string) {
  return useQuery({
    queryKey: ["proposal", id],
    queryFn: () => apiGet<ProposalReview>(`/api/proposals/${id}`),
  });
}

export function useRegression(id: string) {
  return useQuery({
    queryKey: ["regression", id],
    queryFn: () => apiGet<RegressionResult>(`/api/regressions/${id}`),
  });
}

function useGateMutation(path: (id: string) => string, invalidateKey: string) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (vars: { id: string; project: string; extra?: Record<string, unknown> }) =>
      apiPost(path(vars.id), { project: vars.project, ...(vars.extra ?? {}) },
        { idempotencyKey: `${path(vars.id)}:${vars.project}` }),
    onSuccess: (_d, vars) => qc.invalidateQueries({ queryKey: [invalidateKey, vars.id] }),
  });
}

export const useApproveProposal = () =>
  useGateMutation((id) => `/api/proposals/${id}/approve`, "proposal");
export const usePublishRegression = () =>
  useGateMutation((id) => `/api/regressions/${id}/publish`, "regression");
export const useRollbackRegression = () =>
  useGateMutation((id) => `/api/regressions/${id}/rollback`, "regression");
export const useNoSignificantChange = () =>
  useGateMutation((id) => `/api/regressions/${id}/no-significant-change`, "regression");
export const useRevertBaseline = () =>
  useGateMutation((id) => `/api/baselines/${id}/revert`, "baselines");
```

```tsx
// flywheel/ui/src/routes/RegressionReview.tsx
import { useRegression, usePublishRegression, useRollbackRegression,
         useNoSignificantChange } from "../api/hooks";

export function RegressionReview(
  { regressionId, project, candidateFingerprint, canPublish }:
  { regressionId: string; project: string; candidateFingerprint: string; canPublish: boolean },
) {
  const { data: r, isLoading } = useRegression(regressionId);
  const publish = usePublishRegression();
  const rollback = useRollbackRegression();
  const noChange = useNoSignificantChange();
  if (isLoading || !r) return <p>Loading regression…</p>;
  return (
    <section>
      <h1>Regression {r.id}</h1>
      <p>Baseline <code>{r.baseline_fingerprint}</code> vs candidate <code>{r.candidate_fingerprint}</code></p>
      <p>Judge (same for both): <code>{r.judge_version}</code></p>
      <p>Pass-rate delta: {r.pass_rate_delta} (CI {r.pass_rate_ci[0]} … {r.pass_rate_ci[1]})</p>
      <table>
        <thead><tr><th>metric</th><th>expected</th><th>actual</th></tr></thead>
        <tbody>
          {Object.keys(r.expected_metric_delta).map((k) => (
            <tr key={k}><td>{k}</td><td>{r.expected_metric_delta[k]}</td>
              <td>{r.actual_metric_delta[k] ?? "—"}</td></tr>
          ))}
        </tbody>
      </table>
      <p>Fixed failures: {r.fixed_failures.join(", ") || "none"}</p>
      <p>New failures: {r.new_failures.join(", ") || "none"}</p>
      <div role="group" aria-label="regression decision">
        <button disabled={!canPublish}
          onClick={() => publish.mutate({ id: r.id, project,
            extra: { candidate_fingerprint: candidateFingerprint } })}>Publish candidate</button>
        <button onClick={() => rollback.mutate({ id: r.id, project })}>Rollback</button>
        <button onClick={() => noChange.mutate({ id: r.id, project })}>No significant change</button>
      </div>
    </section>
  );
}
```
